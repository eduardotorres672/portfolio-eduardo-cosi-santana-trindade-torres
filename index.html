rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // Default deny match /{document=**} { allow read, write: if false; } // Helper functions function isSignedIn() { return request.auth != null; } function isOwner(userId) { return isSignedIn() && request.auth.uid == userId; } function isValidId(id) { return id is string && id.size() <= 128 && id.matches('^[a-zA-Z0-9_\\-]+$'); } function isValidQRCode(data) { return data.url is string && data.url.size() <= 2048 && data.userId == request.auth.uid && data.config is map && data.createdAt == request.time; } match /qrcodes/{qrcodeId} { allow read: if isSignedIn() && resource.data.userId == request.auth.uid; allow create: if isSignedIn() && isValidId(qrcodeId) && isValidQRCode(request.resource.data); allow update: if isSignedIn() && isOwner(resource.data.userId) && isValidQRCode(request.resource.data) && request.resource.data.userId == resource.data.userId; // Immutable owner allow delete: if isSignedIn() && isOwner(resource.data.userId); } } }
